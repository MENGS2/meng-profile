function updateClock() { var now = new Date(); var hours = String(now.getHours()).padStart(2, '0'); var minutes = String(now.getMinutes()).padStart(2, '0'); var seconds = String(now.getSeconds()).padStart(2, '0'); var clockEl = document.getElementById('digital-clock'); if (clockEl) { clockEl.textContent = hours + ':' + minutes + ':' + seconds; } } setInterval(updateClock, 1000); updateClock(); function createParticle(e) { var clientX = e.clientX || (e.touches && e.touches[0].clientX); var clientY = e.clientY || (e.touches && e.touches[0].clientY); if (!clientX || !clientY) return; var particle = document.createElement('div'); particle.className = 'cursor-particle'; document.body.appendChild(particle); particle.style.left = clientX + 'px'; particle.style.top = clientY + 'px'; var size = Math.random() * 6 + 4; particle.style.width = size + 'px'; particle.style.height = size + 'px'; setTimeout(function() { particle.style.transform = 'translate(-50%, -50%) scale(0)'; particle.style.opacity = '0'; }, 50); setTimeout(function() { particle.remove(); }, 850); } window.addEventListener('mousemove', createParticle); window.addEventListener('touchmove', createParticle);
